
public class Ca {

    // Variables globales
    static int a = 10;
    static int b = 5;
    static double r;

    /**
     * Función de división.
     * No se utiliza porque la división no está permitida.
     *
     * @return resultado de la división
     */
    /*
    static double dv() {
        return (double) a / b;
    }
    */

    // Función de suma
    static void sm() {
        r = a + b;
    }

    // Función de resta
    static void rs() {
        r = a - b;
    }

    // Función de multiplicación
    static void ml() {
        r = a * b;
    }

    // Función de potencia
    static void pt() {
        r = Math.pow(a, b);
    }

    // Función principal
    public static void main(String[] ar) {

        sm();
        System.out.println("Suma: " + r);

        rs();
        System.out.println("Resta: " + r);

        ml();
        System.out.println("Multiplicacion: " + r);

        pt();
        System.out.println("Potencia: " + r);
    }
}
